36: function MiniDealSteps({ status }) {
37:   const steps = [
38:     { key: 'ACCEPTED', label: 'Accepted' },
39:     { key: 'QC_PENDING', label: 'QC Pending' },
40:     { key: 'VERIFIED', label: 'Verified' },

44:   const statusMap = {
45:     ACCEPTED: 0, PHOTO_PENDING: 0, AI_FLAGGED: 0,
46:     AI_REVIEW: 1, AI_PASSED: 1, HUMAN_REVIEW: 1,
47:     VERIFIED: 2,
48:     COMPLETED: 3, DISPUTED: 3,
49:   };
50:   const currentIdx = statusMap[status] ?? -1;
51: 
52:   return (
53:     <div className="space-y-0">
54:       {steps.map((step, idx) => {
55:         const done = idx < currentIdx;

57:         return (
58:           <div key={step.key} className="flex items-start gap-2.5">
59:             <div className="flex flex-col items-center">
60:               <div className={`w-4 h-4 rounded-full flex items-center justify-center shrink-0 ${
61:                 done ? 'bg-red-700' : active ? 'bg-red-600 ring-2 ring-red-200' : 'bg-gray-200'
62:               }`}>
63:                 {done && <svg className="w-2.5 h-2.5 text-white" fill="none" viewBox="0 0 24 24" stroke="currentColor" strokeWidth={3}><path strokeLinecap="round" strokeLinejoin="round" d="M5 13l4 4L19 7" /></svg>}
64:               </div>
65:               {idx < steps.length - 1 && (
66:                 <div className={`w-0.5 h-5 ${done ? 'bg-red-700' : 'bg-gray-200'}`} />
67:               )}
68:             </div>
69:             <span className={`text-xs pt-0.5 ${done ? 'text-red-800 font-bold' : active ? 'text-red-700 font-bold' : 'text-gray-400 font-medium'}`}>
70:               {active && '● '}{step.label}
71:             </span>
72:           </div>
73:         );
74:       })}

76:   );
77: }
